cmake_minimum_required(VERSION 3.16)
project(m1_protocol_tests LANGUAGES C)
enable_testing()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(m1_motion_test m1_motion_test.c ../Core/Src/m1_canopen.c ../Core/Src/m1_motion.c)
target_include_directories(m1_motion_test PRIVATE ../Core/Inc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(m1_motion_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
    target_link_libraries(m1_motion_test PRIVATE m)
endif()
add_test(NAME m1_motion_controller COMMAND m1_motion_test)
add_executable(m1_canopen_test m1_canopen_test.c ../Core/Src/m1_canopen.c)
target_include_directories(m1_canopen_test PRIVATE ../Core/Inc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(m1_canopen_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
    target_link_libraries(m1_canopen_test PRIVATE m)
endif()
add_test(NAME m1_canopen_protocol COMMAND m1_canopen_test)
add_executable(m1_diagnostics_test m1_diagnostics_test.c ../Core/Src/m1_canopen.c ../Core/Src/m1_diagnostics.c)
target_include_directories(m1_diagnostics_test PRIVATE ../Core/Inc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(m1_diagnostics_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
    target_link_libraries(m1_diagnostics_test PRIVATE m)
endif()
add_test(NAME m1_read_only_diagnostics COMMAND m1_diagnostics_test)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME m1_agx_health_contract COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/health_integration.py $<TARGET_FILE:m1_diagnostics_test>)
endif()
