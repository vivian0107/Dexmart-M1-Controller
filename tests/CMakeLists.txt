cmake_minimum_required(VERSION 3.16)
project(m1_protocol_tests LANGUAGES C)
enable_testing()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(m1_canopen_test m1_canopen_test.c ../Core/Src/m1_canopen.c)
target_include_directories(m1_canopen_test PRIVATE ../Core/Inc)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(m1_canopen_test PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
    target_link_libraries(m1_canopen_test PRIVATE m)
endif()
add_test(NAME m1_canopen_protocol COMMAND m1_canopen_test)
